#   uv venv .venv-evilgenie
#   uv pip install --python .venv-evilgenie -r reward_hack_evals/requirements-evilgenie.txt
#   .venv-evilgenie/bin/python scripts/run_reward_hack_evals.py --eval evilgenie ...
#
# EvilGenie itself is vendored (evilgenie/, MIT) and put on sys.path by the runner — nothing to
# install for the package itself. Needs a Docker daemon: its task hardcodes
# sandbox=("docker", Dockerfile).
